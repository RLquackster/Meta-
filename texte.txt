<script>
const enemies=[{name:'Yian Kut-Ku',hp:120,max:120,atk:18,def:5},{name:'Yian Kut-Ku Alpha',hp:180,max:180,atk:25,def:8},{name:'Rathalos',hp:250,max:250,atk:32,def:12}];
let cur=0,e={...enemies[0]},p={hp:150,maxHp:150,mp:50,maxMp:50,atk:28,def:8,pots:3},turn=1,busy=false;
function log(t){const l=document.getElementById('log');l.innerHTML+='<br>> '+t;l.scrollTop=l.scrollHeight;}
function updateBars(){
document.getElementById('eHp').style.width=Math.max(0,e.hp/e.max*100)+'%';
document.getElementById('eHpTxt').innerText=Math.max(0,e.hp)+'/'+e.max;
document.getElementById('pHp').style.width=p.hp/p.maxHp*100+'%';
document.getElementById('pHpTxt').innerText=p.hp+'/'+p.maxHp;
document.getElementById('pMp').style.width=p.mp/p.maxMp*100+'%';
document.getElementById('pMpTxt').innerText=p.mp+'/'+p.maxMp;
document.getElementById('potCount').innerText=p.pots;
document.getElementById('turn').innerText='TOUR '+turn;
}
function dmgPopup(el,val,color='#ff3333'){
const s=document.getElementById('scene');const d=document.createElement('div');d.className='dmg';d.style.color=color;d.innerText=val>0?'-'+val:'+'+Math.abs(val);
const r=el.getBoundingClientRect();const sr=s.getBoundingClientRect();
d.style.left=(r.left-sr.left+r.width/2)+'px';d.style.top=(r.top-sr.top)+'px';
s.appendChild(d);setTimeout(()=>d.remove(),800);
}
function doAction(type){
if(busy) return; busy=true; let dmg=0;
if(type==='attack'){dmg=Math.max(1,p.atk+Math.floor(Math.random()*8)-e.def);e.hp-=dmg;log(`Tu frappes ! -${dmg} PV`);hit('enemy',dmg);}
else if(type==='magic'){if(p.mp<10){log('Pas assez de mana !');busy=false;return;}p.mp-=10;dmg=p.atk+15+Math.floor(Math.random()*12);e.hp-=dmg;log(`Boule de feu ! 🔥 -${dmg} PV`);hit('enemy',dmg,'#ff8844');}
else if(type==='heal'){if(p.mp<15){log('Pas assez de mana !');busy=false;return;}p.mp-=15;let heal=40+Math.floor(Math.random()*15);p.hp=Math.min(p.maxHp,p.hp+heal);log(`Soin ! +${heal} PV`);dmgPopup(document.getElementById('player'),-heal,'#44ff44');updateBars();setTimeout(enemyTurn,800);return;}
else if(type==='item'){if(p.pots<=0){log('Plus de potions !');busy=false;return;}p.pots--;let heal=60;p.hp=Math.min(p.maxHp,p.hp+heal);log(`Potion ! +${heal} PV`);dmgPopup(document.getElementById('player'),-heal,'#44ff44');updateBars();setTimeout(enemyTurn,800);return;}
updateBars();if(e.hp<=0){victory();return;}setTimeout(enemyTurn,900);
}
function hit(who,dmg,col){const el=document.getElementById(who);el.classList.add('hit');dmgPopup(el,dmg,col||'#ff3333');setTimeout(()=>el.classList.remove('hit'),200);}
function enemyTurn(){turn++;let dmg=Math.max(1,e.atk+Math.floor(Math.random()*10)-p.def);p.hp-=dmg;log(`${e.name} attaque ! -${dmg} PV`);hit('player',dmg);updateBars();if(p.hp<=0){document.getElementById('vicText').innerText='DÉFAITE...\nRecommencer ?';document.getElementById('victory').classList.add('show');}busy=false;}
function victory(){document.getElementById('vicText').innerText=`VICTOIRE !\n${e.name} vaincu !`;document.getElementById('victory').classList.add('show');busy=false;}
function nextBattle(){document.getElementById('victory').classList.remove('show');cur++;if(cur>=enemies.length)cur=0;e={...enemies[cur]};turn=1;p.mp=Math.min(p.maxMp,p.mp+20);log(`---\nUn ${e.name} apparaît ! HP:${e.max}`);updateBars();}
updateBars();
